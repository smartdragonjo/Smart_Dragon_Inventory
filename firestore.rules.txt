rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isOwner() {
      return signedIn()
        && request.auth.token.email == 'smartdragonjordan@gmail.com';
    }

    function isEditor() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.enabled == true
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'editor';
    }

    function isAdminUser() {
      return isOwner() || isEditor();
    }

    function hasEnabledInvite() {
      return signedIn()
        && request.auth.token.email != null
        && exists(/databases/$(database)/documents/user_invites/$(request.auth.token.email))
        && get(/databases/$(database)/documents/user_invites/$(request.auth.token.email)).data.enabled == true
        && get(/databases/$(database)/documents/user_invites/$(request.auth.token.email)).data.role == 'editor';
    }

    match /vehicles/{vehicleId} {
      allow read: if resource.data.status == 'approved' || isAdminUser();
      allow create, update, delete: if isOwner();
    }

    match /vehicle_fitments/{fitmentId} {
      allow read: if isAdminUser()
        || (
          resource.data.status == 'approved'
          && exists(/databases/$(database)/documents/vehicles/$(resource.data.vehicleId))
          && get(/databases/$(database)/documents/vehicles/$(resource.data.vehicleId)).data.status == 'approved'
        );
      allow create, update, delete: if isOwner();
    }

    match /change_requests/{requestId} {
      allow create: if isAdminUser()
        && request.resource.data.createdByUid == request.auth.uid
        && request.resource.data.status == 'pending_review';
      allow read: if isOwner()
        || (isEditor() && resource.data.createdByUid == request.auth.uid);
      allow update, delete: if isOwner();
    }

    match /categories/{categoryId} {
      allow read: if resource.data.active == true || isAdminUser();
      allow create, update, delete: if isOwner();
    }

    match /user_invites/{inviteEmail} {
      allow read: if isOwner()
        || (signedIn() && request.auth.token.email == inviteEmail);
      allow create, update, delete: if isOwner();
    }

    match /users/{userId} {
      allow read: if isOwner() || (signedIn() && request.auth.uid == userId);

      allow create: if isOwner()
        || (
          signedIn()
          && request.auth.uid == userId
          && request.auth.token.email != null
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.email == request.auth.token.email
          && request.resource.data.role == 'editor'
          && request.resource.data.enabled == true
          && hasEnabledInvite()
        );

      allow update, delete: if isOwner();
    }

    match /audit_logs/{logId} {
      allow read, create: if isOwner();
      allow update, delete: if false;
    }

    match /backups/{backupId} {
      allow read, create, update, delete: if isOwner();
    }

    match /cars/{carId} {
      allow read: if true;
      allow write: if signedIn()
        && request.auth.token.email in [
          'smartdragonjordan@gmail.com',
          'forcesmahdi@gmail.com'
        ];
    }
  }
}
